const createResponce = ( no, cnt = 0, idCheck = false ) => {

    let element = globalBoard[no];
    
    if( no > globalBoard.length ){
        return false;
    }    
    if( element['del'] ){
        return false;
    }
    if(idCheck){
        if( element['id'] != idCheck || element['admin'] || element['author'] ){
            return false;
        }
    }else{
        if( gameNo === 2 ){
            
        }else if ( element['gameNo'] != gameNo ){
            return false;
        }
    }

    cnt++;
    
    let nameArea = '';
    let idArea = '<a style="visibility:hidden">id:A000000<a>';
    let comment = element['comment'].replace(/&gt;&gt;[0-9]{1,}/g, (s) => {
        let responceNo = s.replace(/&gt;&gt;/g, '');
        return `<a class="text-success responceChecker" data-no="${responceNo}">${s}</a> `;
    });
    
    let gameTitle = ( gameNo === 2 || idCheck ) ? `(${globalDataes[ element['gameNo'] ]['title']})` : '' ;
    
    if(element['admin']){
        nameArea = `<b class="text-danger">${no}: ${element['writer']}</b>`;
    }else if(element['author']){
        nameArea = `<b class="text-success">${no}: ${element['writer']}</b>`;
    }else{
        nameArea = `<a class="text-primary">${no}: ${element['writer']}</a>`;
        idArea = `<a class="text-success idChecker" data-id="${element['id']}">id:${element['id']}</a>`;
    }
    
    str = `${nameArea}<br>
        ${comment}<br>
        <div class="text-right text-secondary">
        No.${element['gameNo']}&nbsp;
        ${gameTitle}&nbsp;&nbsp;
        ${element['time']}&nbsp;&nbsp;&nbsp;
        ${idArea}
        </div>
        <hr>`;
    return [str, cnt];
}


$('body').on( 'click', '.responceChecker', function() {
    let no = $(this).data('no');
    let responce = createResponce(no);
    
    if(responce){
        $('#modalBoardBody').html(responce[0]);
        $('#modalBoard').modal('show');
    }
});


$('body').on( 'click', '.idChecker', function() {

const getGoogleUrl = (googleId) => {
    if(!googleId){
        return '';
    }
    return 'https://drive.google.com/uc?export=view&id=' + googleId;
}



const updateInformation = () => {
    let element = globalDataes[gameNo];
    let title = `<img src="${getGoogleUrl(element['icon'])}"> No.${gameNo}: ${element['title']}`;

    let ss1 = getGoogleUrl(element['ss1']);
    let ss2 = getGoogleUrl(element['ss2']);

    let image;
    if(ss2){
        image = `<img class="img-fluid" src="${ss1}"
        onmouseover="this.src='${ss2}'"
        onmouseout="this.src='${ss1}'">
        <img src="${ss2}" style="display:none">
        `;
    }else{
        image = `<img class="img-fluid" src="${ss1}">`;
    }

    let body = `
        ${title}
        <hr>
        ${image}
        <hr>
        作者名: ${element['author']}<br>
        ツール: ${ element['tkool'] === 'other' ? element['tkool_other'] : element['tkool'] }<br>
        動画や配信: ${ element['video'] === 'other' ? element['video_other'] : element['video'] }<br>
        ジャンル: ${element['genre']}<br>
        <hr>
        作者コメント:<br>
        ${element['comment_author']}<br>
        `;
    $('#gameInformation').html(body);
    $('#displayNo').text(gameNo + ' ' + element['title'] );
    $('#urlViewer').val('https:\/\/script.google.com\/macros\/s\/AKfycbxUhsUdhTG6F844hBnioDdGacKRTip815r48sgbP-pPUhoXuPsK\/exec?game=' + gameNo);
    
    
    
    
    if( element['game'] == 'false' ){
        $('#downloadGame').hide();
        $('#lastUpdateDisplay').text('');
    }else{
        $('#downloadGame').show();
        $('#downloadGame').attr( 'href', 'https://drive.google.com/file/d/' + element['game'] + '/view?usp=drivesdk' );
        
        let date = new Date(element['spare2']);
        let updateDate = ( date.getMonth() + 1 ) + "/" + date.getDate();
        $('#lastUpdateDisplay').text(updateDate);
        
    }
    if( element['patch'] == 'false' ){
        $('#downloadPatch').hide();
    }else{
        $('#downloadPatch').show();
        $('#downloadPatch').attr( 'href', 'https://drive.google.com/file/d/' + element['patch'] + '/view?usp=drivesdk' );
    }
    if( element['spare3'] ){
        $('#downloadGame').text( Math.round( element['spare3'] / 1024/ 1024 * 10 ) / 10 + 'MB' );
    }
    
    let dlCount = ( element['board'] != 'false' ) ? element['board'] : 0;
    dlCount = ( element['spare1'] == 'public' ) ? dlCount : '非公開';
    dlCount = ( element['game'] != 'false' ) ? dlCount : '';
    $('#downloadCount').text( dlCount );

}


const preloadingImage = () => {
    let str = '';
    for( let i = 0; i < globalDataes.length; i++ ){
        
        let element = globalDataes[i];
        
        let icon = ( element['icon'] == '' ? '' : `<img src="${getGoogleUrl(element['icon'])}">` );
        let ss1 = ( element['ss1'] == '' ? '' : `<img src="${getGoogleUrl(element['ss1'])}">` );
        let ss2 = ( element['ss2'] == '' ? '' : `<img src="${getGoogleUrl(element['ss2'])}">` );
        
        str += icon + ss1 + ss2;
    }
    $('#preLoadingImage').html(str);
}




const updateBoard = () => {
    let str = '';
    let cnt = 0;
    
    for( let i = 0; i < globalBoard.length; i++ ){
        let responce = createResponce(i, cnt);
        
        if(responce){
            str += responce[0];
            cnt = responce[1];
        }
    }
    $('#boardArea').html(str);
    
    let cntComment = [];
    let lastComment = [];
    for( let n = 0; n < globalDataes.length; n++ ){
        cntComment[n] = 0;
        lastComment[n] = '';
    }
    
    for( let i = 0; i < globalBoard.length; i++ ){
        let element = globalBoard[i];
        if(element['del']){
            continue;
        }
        if( isNaN(element['gameNo']) ){
            continue;
        }
        let n = element['gameNo'];
        cntComment[n] = cntComment[n] + 1;
        lastComment[n] = element['time'].substr(3, 11);
        lastComment[2] = element['time'].substr(3, 11);
    }

    
    for( let n = 0; n < globalDataes.length; n++ ){
        let target = '#lastComment' + n;
        $(target).text(lastComment[n]);
    }
        

}

const afterUpload = (dataes) => {
    const reWrite = () => {
        $('#submitBtn').prop('disabled', false);
        $('#information').text('再書き込みが可能になりました');   
    }
    
    if(dataes){
        
        globalBoard = dataes;
        updateBoard();
        alert('書き込みが終了しました');

        $('#writer').val('');
        $('#comment').val('');

        
        let count = 20;
        const timer = setInterval( () => {
            count--;
            $('#information').text(`再度書き込み可能になるまで${count}秒ほどお待ち下さい`);

            if ( count <= 0 ){
                clearInterval(timer);
                $('#submitBtn').prop('disabled', false);
                $('#information').text('再書き込みが可能になりました');                           
            }
        }, 1000);

    }else{
        $('#information').text('');
        $('#submitBtn').prop('disabled', false);
        alert('かきこみ失敗');
    }
}

const gameCountDown = () => {

$(window).on( 'keydown', (e) => {

    let focused = $(':focus');        
    
    if( focused.hasClass('form-control') || focused.hasClass('btnGroup') ){
        return;
    }
    
    
    if( e.keyCode === 37 || e.keyCode === 52 ){
        gameCountDown();
    }
    if( e.keyCode === 39 || e.keyCode === 54 ){
        gameCountUp();
    }
    
    
    if( e.keyCode === 48 ){
        let modal = $('#modal');
        if( modal.hasClass('show') ){
            modal.modal('hide');
        }else{
            modal.modal('show');
        }
    }
    
    
    if( e.keyCode === 56 ){
        let modal = $('#modal');
        $('#movePositionTop').trigger('click');
    }
    if( e.keyCode === 53 ){
        let modal = $('#modal');
        $('#movePositionMiddle').trigger('click');
    }
    if( e.keyCode === 50 ){
        let modal = $('#modal');
        $('#movePositionBottom').trigger('click');
    }
});
